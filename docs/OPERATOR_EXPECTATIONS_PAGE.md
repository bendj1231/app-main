```typescript
interface OperatorExpectationsPage {
  operator_id: string;
  
  // Branding
  company_name: string;
  tagline: string;
  since_year: number;

  compensation: {

  };
  
  crew_treatment: {
    positioning_class: 'economy' | 'premium_economy' | 'business' | 'first';

    hotel_tier: number; // 3, 4, 5 star
    hotel_occupancy: 'shared' | 'single';
    layover_allowance: number;
    crew_rest_minimum_hours: number;
  };
  
  operational_support: {
    legal_support_24_7: boolean;
    countries_covered: number;
    customs_pre_clearance: boolean;
    dedicated_controller: boolean;
    medical_evacuation: boolean;
    security_detail: boolean;
  };
  
  // Competitive differentiation
  unique_selling_points: string[];
  operator_testimonials: Testimonial[];
  
  // Active pathways
  open_pathways: PathwaySummary[];
  
  // Engagement metrics
  total_alignments: number;
  average_response_time_hours: number;
  pilot_rating: number; // 1-5 stars
```

### The Alignment Mechanic

```typescript
const alignProfileToOperator = async (
  pilotId: string,
  operatorId: string
) => {
  // 1. Record the alignment
  const alignment = await createAlignment({
    pilot_id: pilotId,
    operator_id: operatorId,
    aligned_at: new Date().toISOString(),
    pilot_visible: true,
    operator_visible: true // Now operator sees pilot's verified data
  });
  
  // 2. Notify operator
  await notifyOperator(operatorId, {
    type: 'new_alignment',
    message: 'A pilot has aligned their profile with your company',
    pilot_tier: 'recognition_plus',
    pilot_summary: '6,200 PIC hours, B737/A320 rated, Veremark verified'
  });
  
  // 3. Notify pilot
  await notifyPilot(pilotId, {
    type: 'alignment_confirmed',
    message: `Your profile is now visible to ${operatorName}`,
    next_step: 'Wait for operator to poke you, or browse other operators'
  });
};
```
